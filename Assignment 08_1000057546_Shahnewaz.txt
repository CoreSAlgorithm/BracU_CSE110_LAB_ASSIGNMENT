import java.util.Scanner;
public class HomeTask08
{
  public static void main(String [] args)
  {
    int result = special_sum(8);
    System.out.println(result);
  }
   public static boolean isPrime(int a)
  {
    if(a<=1)
    {
      return false;
    }
    for(int i=2; i<=a/2; i++)
    {
      if(a%i==0)
      {
        return false;
      }
    }
    return true;
  }
   public static boolean isPerfect(int a)
   {
     if(a<=1)
     {
       return false;
     }
     int sum = 0;
       for(int i=1; i<=a/2; i++)
       {
         if(a%i == 0)
         {
           sum+=i;
         }
       }
     return sum == a;
   }
  public static int special_sum(int a)
  {
    int sum = 0;
    for(int i=1; i<=a; i++)
    {
      boolean check1 = isPrime(i);
      if(check1==true)
      {
        sum+=i;
      }
    }
    int sum2 = 0;
    for(int i=1; i<=a; i++)
    {
      boolean check2 = isPerfect(i);
      if(check2 == true)
      {
        sum2+=i;
      }
    }
    return sum+sum2;
  }
}

import java.util.Scanner;
public class HomeTask08
{
  public static void main(String [] args)
  {
    showDiamond(5);
  }
  public static void showDots(int a)
  {
    for(int i=0; i<a; i++)
    {
      System.out.print(".");
    }
    System.out.print("\n");
  }
  public static void show_palidrome(int a)
  {
    for (int i=1; i<=a; i++)
    {
      System.out.print(i);
    }
    for(int k = a-1; k>=1; k--)
    {
      System.out.print(k);
    }
    System.out.print("\n");
  }
  public static void showDiamond(int n)
  {
    for (int i=1; i<=n; i++)
    {
      for(int j = i; j<n; j++)
      {
        System.out.print(".");
      }
      for(int k = 1; k<=i; k++)
      {
        System.out.print(k);
      }
      for(int k = i-1; k>=1; k--)
      {
        System.out.print(k);
      }
      for(int j = i; j<n; j++)
      {
        System.out.print(".");
      }
      System.out.println();
    }
    for(int i = n-1; i>0; i--)
    {
      for(int j = n-i; j>0; j--)
      {
        System.out.print(".");
      }
      for(int k = 1; k<=i; k++)
      {
        System.out.print(k);
      }
      for(int k = i-1; k>=1; k--)
      {
        System.out.print(k);
      }
      for(int j = n-i; j>0; j--)
      {
        System.out.print(".");
      }
      System.out.println();
    }
  }
}

import java.util.Scanner;
public class HomeTask08
{
  public static void main(String [] args)
  {
    Scanner sc = new Scanner(System.in);
    System.out.println("Enter your age:");
    int a = sc.nextInt();
    double total = 0;
    for(int i=1; i<=12; i++)
    {
      System.out.print("Monthly income:");
      int b = sc.nextInt();
      double t2 = calcYearlyTax(a,b);
      total += t2;
      System.out.printf("Month"+i+" tax:%.2f",t2);
      System.out.println();
    }
    System.out.printf("Total yearly tax:%.2f",total);
    System.out.println();
  }
  public static double calcTax(int a,int b)
  {
    double c = 0.0;
    if(a<18 || b<10000)
    {
      return c;
    }
    else if(a>18 && (b>=10000 && b<=20000))
    {
      c = b*0.07;
      return c;
    }
    else
    {
      c = b*0.14;
    }
    return c;
  }
  public static double calcYearlyTax(int a,int b)
  {
    double d = calcTax(a,b);
    return d;
  }
}
