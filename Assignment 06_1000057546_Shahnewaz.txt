      do
      {
        temp = sc.next();

          max = temp.length();
          maxStr = temp;
        }
      } 
      while (!temp.equals("stop"));
        int power = 0;
        temp = "";
        for (int i = 0; i < maxStr.length(); i++)
        {
          char ch = maxStr.charAt(i);
          if (ch >= 'a' && ch <= 'z')
          {
            ch -= 32;
            temp += ch;
            power += (ch);
          } 
          else if (ch >= 'A' && ch <= 'Z')

            power += ch;
          }
        }
